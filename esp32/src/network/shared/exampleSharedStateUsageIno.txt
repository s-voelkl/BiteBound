// Threading Synchronizations
SemaphoreHandle_t stateMutex = nullptr;
QueueHandle_t commandQueue = nullptr;
SharedStateData sharedState;

// Global Displays
Arduino_DataBus *bus = new Arduino_ESP32SPI(pin_lcd_dc, pin_lcd_cs, pin_lcd_sck, pin_lcd_mosi);
Arduino_GFX *gfx = new Arduino_ST7789(bus, pin_lcd_rst, 0, true, display_width, display_height, 0, 20, 0, 0);
GraphicsManager graphicsManager(display_width, display_height);

const int play_height = display_height - ui_header_height;

void vNetworkTask(void *pvParameters);
void vGameTask(void *pvParameters);

/**
 * @brief MQTT Message handler callback to queue command payloads safely.
 */
void handleMqttMessage(char* topic, byte* payload, unsigned int length) {
    char* cleanPayload = new char[length + 1];
    memcpy(cleanPayload, payload, length);
    cleanPayload[length] = '\0';

    if (strcmp(topic, mqtt_command_topic) == 0) {
        CommandMsg cmdMsg;
        if (CommandParser::parse(cleanPayload, cmdMsg)) {
            if (xQueueSend(commandQueue, &cmdMsg, 0) != pdPASS) {
                Serial.println("Queue Full: Command rejected.");
            }
        }
    }
    delete[] cleanPayload;
}

void setup() {
    Serial.begin(115200);
    Serial.println("Initializing Multithreaded BiteBound Firmware...");

    // Create RTOS primitives
    stateMutex = xSemaphoreCreateMutex();
    commandQueue = xQueueCreate(cmd_queue_length, cmd_queue_item_size);
    initSharedState(sharedState);

#if RUN_TESTS
    delay(3000);
    Serial.println("Running test suites...");
#else
    if (!gfx->begin()) {
        Serial.println("Hardware Error: GFX Display Failure.");
    }
    pinMode(pin_lcd_bl, OUTPUT);
    digitalWrite(pin_lcd_bl, HIGH);
    graphicsManager.begin(gfx);
    graphicsManager.drawLoadingScreen("System Launching...");

    // Spawn pinned concurrent loops
    xTaskCreatePinnedToCore(
        vNetworkTask,
        "NetworkTask",
        8192,
        NULL,
        1,
        NULL,
        core_network
    );

    xTaskCreatePinnedToCore(
        vGameTask,
        "GameTask",
        8192,
        NULL,
        2, // Higher priority for game ticks stability
        NULL,
        core_game
    );
#endif
}

void loop() {
#if RUN_TESTS
    aunit::TestRunner::setTimeout(30);
    aunit::TestRunner::run();
#else
    // Keep standard loop idle to prevent scheduling starvation of core 1
    vTaskDelay(pdMS_TO_TICKS(1000));
#endif
}

void vNetworkTask(void *pvParameters) {
    Serial.print("Network Task pinned to Core: ");
    Serial.println(xPortGetCoreID());

    wifiManager.connect();
    timeManager.sync();
    mqttManager.begin();
    
    // Wire up standard callback implementation
    // Assuming underlying PubSubClient registration inside MqttManager or direct mapping
    // mqttManager.setCallback(handleMqttMessage);

    mqttManager.subscribe(mqtt_command_topic, 1);
    TickType_t lastTelemetryTime = xTaskGetTickCount();

    while (true) {
        mqttManager.connect();
        mqttManager.loop();

        TickType_t now = xTaskGetTickCount();
        if ((now - lastTelemetryTime) >= pdMS_TO_TICKS(telemetry_rate_ms)) {
            lastTelemetryTime = now;
            TelemetryData telemetry;

            // Fetch state with low contention
            {
                MutexLock lock(stateMutex);
                if (lock.isLocked()) {
                    telemetry.client_id = device_id;
                    telemetry.hardware = device_hardware;
                    telemetry.firmware_version = device_firmware_version;
                    telemetry.uptime_ms = millis();
                    telemetry.wifi_ssid = wifiManager.getSSID();

                    telemetry.game_id = sharedState.gameId;
                    telemetry.player_name = sharedState.playerName;
                    telemetry.target_cookies = sharedState.cookiesCount;
                    telemetry.screen_width = display_width;
                    telemetry.screen_height = display_height;
                    telemetry.wall_thickness_px = sharedState.wallThicknessPx;

                    telemetry.status = sharedState.isRunning ? "running" : "idle";
                    telemetry.cookies_collected = sharedState.cookiesCollected;
                    telemetry.cookies_remaining = sharedState.cookiesRemaining;
                    telemetry.current_round = sharedState.currentRound;
                    telemetry.elapsed_time_sec = sharedState.elapsedTimeSec;

                    telemetry.ball_pos_x = sharedState.ballPosX;
                    telemetry.ball_pos_y = sharedState.ballPosY;
                    telemetry.velocity_x = sharedState.velocityX;
                    telemetry.velocity_y = sharedState.velocityY;
                    telemetry.acc_x = sharedState.accX;
                    telemetry.acc_y = sharedState.accY;
                }
            }

            SensorData sensorData = MOCK_SENSORS ? sensorManager.readMock() : sensorManager.read();
            telemetry.accel_x = sensorData.accelerometerX;
            telemetry.accel_y = sensorData.accelerometerY;
            telemetry.accel_z = sensorData.accelerometerZ;
            telemetry.gyro_x = sensorData.gyroscopeX;
            telemetry.gyro_y = sensorData.gyroscopeY;
            telemetry.gyro_z = sensorData.gyroscopeZ;
            telemetry.battery_voltage = sensorData.batteryVoltage;
            telemetry.button = sensorData.button;

            String payload = buildTelemetryJson(telemetry);
            mqttManager.publish(mqtt_telemetry_topic, payload.c_str(), mqtt_retain);
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

void vGameTask(void *pvParameters) {
    Serial.print("Game Task pinned to Core: ");
    Serial.println(xPortGetCoreID());

    sensorManager.begin();

    PhysicsEngine physicsEngine;
    PhysicsBody ball{ (float)play_width / 2.0f, (float)play_height / 2.0f, 4.0f };
    BorderCollider world(play_width, play_height);

    uint8_t* localGameBoard = new uint8_t[play_width * play_height];
    mazeManager.generate(localGameBoard);

    RectCookieSpawner cookieSpawner(play_width, play_height, 3.0f);
    CookieField cookieField;

    bool active = false;
    int localCookiesCount = 10;
    int localWallThickness = 6;
    unsigned long gameStartTime = 0;

    TickType_t lastTick = xTaskGetTickCount();

    while (true) {
        CommandMsg cmd;
        if (xQueueReceive(commandQueue, &cmd, 0) == pdPASS) {
            MutexLock lock(stateMutex);
            if (lock.isLocked()) {
                if (cmd.type == CommandType::START) {
                    sharedState.isRunning = true;
                    strncpy(sharedState.playerName, cmd.playerName, sizeof(sharedState.playerName) - 1);
                    sharedState.cookiesCount = cmd.cookiesCount;
                    sharedState.wallThicknessPx = cmd.wallThicknessPx;
                    
                    localCookiesCount = cmd.cookiesCount;
                    localWallThickness = cmd.wallThicknessPx;
                    active = true;
                    gameStartTime = millis();

                    ball.x = play_width / 2.0f;
                    ball.y = play_height / 2.0f;
                    ball.vx = 0.0f;
                    ball.vy = 0.0f;

                    PhysicsParams params;
                    params.sensitivity = cmd.imuSensitivity;
                    params.restitution = cmd.bounceRestitution;
                    params.emaAlpha = cmd.emaAlpha;
                    params.deadzone = cmd.deadzoneThreshold;
                    physicsEngine.setParams(params);

                    mazeManager = MazeManager(play_width, play_height, localWallThickness);
                    mazeManager.generate(localGameBoard);
                    cookieField.start(3, localCookiesCount, cookieSpawner, ball);

                    sharedState.cookiesCollected = 0;
                    sharedState.cookiesRemaining = localCookiesCount;
                    sharedState.currentRound = 1;
                    sharedState.elapsedTimeSec = 0.0f;

                    graphicsManager.forceFullRedraw();
                } else if (cmd.type == CommandType::STOP) {
                    sharedState.isRunning = false;
                    active = false;
                } else if (cmd.type == CommandType::PARAM_CHANGE) {
                    PhysicsParams params;
                    params.sensitivity = cmd.imuSensitivity;
                    params.restitution = cmd.bounceRestitution;
                    params.emaAlpha = cmd.emaAlpha;
                    params.deadzone = cmd.deadzoneThreshold;
                    physicsEngine.setParams(params);
                }
            }
        }

        if (active) {
            SensorData s = MOCK_SENSORS ? sensorManager.readMock() : sensorManager.read();
            Vec2 acceleration = physicsEngine.inputAccel(s.accelerometerX, s.accelerometerY);
            physicsEngine.step(ball, acceleration, (float)game_tick_rate_ms / 1000.0f, world);

            cookieField.checkPickup(ball);

            GameState visualState;
            visualState.status = "running";
            visualState.cookiesCollected = cookieField.collected();
            visualState.cookiesRemaining = cookieField.remaining();
            visualState.currentRound = 1;
            visualState.elapsedTimeSec = (float)(millis() - gameStartTime) / 1000.0f;

            graphicsManager.update(
                localGameBoard,
                play_width,
                play_height,
                ball,
                &cookieField.at(0),
                cookieField.count(),
                visualState
            );

            {
                MutexLock lock(stateMutex);
                if (lock.isLocked()) {
                    sharedState.ballPosX = ball.x;
                    sharedState.ballPosY = ball.y;
                    sharedState.velocityX = ball.vx;
                    sharedState.velocityY = ball.vy;
                    sharedState.accX = acceleration.x;
                    sharedState.accY = acceleration.y;
                    sharedState.cookiesCollected = visualState.cookiesCollected;
                    sharedState.cookiesRemaining = visualState.cookiesRemaining;
                    sharedState.elapsedTimeSec = visualState.elapsedTimeSec;
                }
            }
        }
        vTaskDelayUntil(&lastTick, pdMS_TO_TICKS(game_tick_rate_ms));
    }
}
